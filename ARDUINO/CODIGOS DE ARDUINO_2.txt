// CODIGO INICIAL

#include <Wire.h>
#include <LiquidCrystal_I2C.h>

LiquidCrystal_I2C lcd(0x27, 16, 2);

const int led1 = 6;
const int led2 = 7;
const int button = 8;
const int buzzer = 9;

const int repeats = 4;
const int durationSeconds = 10;

// 🔊 SONIDO INICIO
void soundStart() {
  tone(buzzer, 800); delay(120);
  noTone(buzzer); delay(50);

  tone(buzzer, 1000); delay(120);
  noTone(buzzer); delay(50);

  tone(buzzer, 1200); delay(150);
  noTone(buzzer);
}

// 🔊 SONIDO FINAL
void soundEnd() {
  tone(buzzer, 523); delay(150);
  tone(buzzer, 659); delay(150);
  tone(buzzer, 784); delay(200);
  noTone(buzzer);
}

// 📊 PROGRESO
void mostrar(String texto, int sec, int total) {

  int percent = (sec * 100) / total;
  int bars = map(sec, 0, total, 0, 16);

  lcd.clear();

  lcd.setCursor(0, 0);
  lcd.print(texto);

  lcd.setCursor(10, 0);
  lcd.print(percent);
  lcd.print("%");

  lcd.setCursor(0, 1);

  for (int i = 0; i < 16; i++) {
    if (i < bars) lcd.write(byte(255));
    else lcd.print("-");
  }
}

// ⏱ ETAPA
void etapa(String texto, int led) {

  digitalWrite(led1, led == led1);
  digitalWrite(led2, led == led2);

  soundStart();

  for (int s = 1; s <= durationSeconds; s++) {
    mostrar(texto, s, durationSeconds);
    delay(1000);
  }
}

void setup() {

  lcd.init();
  lcd.backlight();

  pinMode(led1, OUTPUT);
  pinMode(led2, OUTPUT);
  pinMode(buzzer, OUTPUT);
  pinMode(button, INPUT_PULLUP);

  lcd.setCursor(0, 0);
  lcd.print("PRESIONA");
  lcd.setCursor(0, 1);
  lcd.print("PARA EMPEZAR (:");

  // 🔥 NO AUTO START
  while (digitalRead(button) == HIGH) {
    delay(10);
  }

  while (digitalRead(button) == LOW) {
    delay(10);
  }
}

void loop() {

  for (int i = 0; i < repeats; i++) {

    etapa("Estudio", led1);

    if (i == repeats - 1) {
      etapa("Desc largo", led2);
    } else {
      etapa("Desc corto", led2);
    }
  }

  // 🏁 FINAL
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("BIEN HECHO");
  lcd.setCursor(0, 1);
  lcd.print("FELICITACIONES");

  soundEnd();

  delay(2500);

  // 🔥 BLOQUEO TOTAL DE REINICIO (FIX FINAL REAL)

  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("PRESIONA");
  lcd.setCursor(0, 1);
  lcd.print("PARA EMPEZAR");

  // esperar que suelte botón primero
  while (digitalRead(button) == LOW) {
    delay(10);
  }

  // esperar nueva presión real
  while (digitalRead(button) == HIGH) {
    delay(10);
  }

  delay(200); // anti rebote

  // 🔥 reinicio real del programa
  return;
}

#include <Wire.h>
#include <LiquidCrystal_I2C.h>

LiquidCrystal_I2C lcd(0x27, 16, 2);

const int soundSensor = 8;

void setup() {
  lcd.init();
  lcd.backlight();
  pinMode(soundSensor, INPUT);

  lcd.setCursor(0, 0);
  lcd.print("Prueba Sensor");
  lcd.setCursor(0, 1);
  lcd.print("Esperando...");
  delay(1000);
}

void loop() {
  int estado = digitalRead(soundSensor);

  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("Estado:");

  if (estado == HIGH) {
    lcd.setCursor(0, 1);
    lcd.print("APLAUSO!");
  } else {
    lcd.setCursor(0, 1);
    lcd.print("Silencio");
  }

  delay(200); // pequeño retardo para estabilidad
}

const int led1 = 6;
const int led2 = 7;

const int buzzer = 9;

const int repeats = 4;
const int durationSeconds = 10;

// Umbrales separados
int umbralInicio = 522;

  tone(buzzer, 1000); delay(120);

  noTone(buzzer);

// 👂 DETECTAR APLAUSO (máximo en 200 ms con umbral configurable)
bool detectarAplauso(int umbral) {
  int maxValor = 0;
  unsigned long inicio = millis();

  // lee durante 200 ms y guarda el máximo
  while (millis() - inicio < 200) {
    int valor = analogRead(soundSensorAO);
    if (valor > maxValor) {
      maxValor = valor;
    }
  }

  Serial.print("Max: "); Serial.print(maxValor);
  Serial.print(" / Umbral: "); Serial.println(umbral);

  // lcd.clear();
  // lcd.setCursor(0, 0);
  // lcd.print("Max:");
  // lcd.print(maxValor);
  // lcd.setCursor(0, 1);
  // lcd.print("Umbral:");
  // lcd.print(umbral);

  if (maxValor > umbral) {
    return true;
  }
  return false;
}

// 🔄 REINICIO REAL
void reiniciarArduino() {
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("REINICIANDO...");
  delay(500);
  wdt_enable(WDTO_15MS); // watchdog en 15 ms
  while (true) {
    // espera a que ocurra el reset
  }
}

// ⏱ ETAPA
void etapa(String texto, int led) {
  digitalWrite(led1, led == led1);
  digitalWrite(led2, led == led2);

  soundStart(); // 🔊 suena al empezar cada etapa

  for (int s = 1; s <= durationSeconds; s++) {
    mostrar(texto, s, durationSeconds);

    // 👀 Si detecta aplauso durante la etapa, reinicia el Arduino
    if (detectarAplauso(umbralReinicio)) {
      reiniciarArduino();
    }

    delay(1000);
  }
}

void setup() {
  lcd.init();
  lcd.backlight();
  Serial.begin(9600);

  pinMode(led1, OUTPUT);
  pinMode(led2, OUTPUT);

  lcd.print("PARA EMPEZAR (:");

  // 🔥 Espera aplauso para iniciar
  while (!detectarAplauso(umbralInicio)) {
    delay(10);
  }
  soundStart();
}

  lcd.print("PARA REINICIAR");
